function playVideo(rawStreamUrl, buttonElement) { document.querySelectorAll('.ep-btn').forEach(btn => btn.classList.remove('active')); if (buttonElement) buttonElement.classList.add('active'); if (currentHls) { currentHls.destroy(); currentHls = null; } let finalStreamUrl = rawStreamUrl; // ถ้าลิงก์ลงท้ายด้วย .m3u8 แบบ Master Playlist ให้ตัดนามสกุลออกแล้วต่อท้ายด้วยไฟล์ index.m3u8 ของความละเอียดจริง if (finalStreamUrl.includes('/newplaylist/') || finalStreamUrl.endsWith('.m3u8')) { // ตัด .m3u8 ตัวท้ายออก แล้วบังคับชี้ไปที่ไฟล์สตรีมความละเอียดสูงสุด let cleanBase = finalStreamUrl.replace(/\/[^\/]+\.m3u8$/, ''); finalStreamUrl = cleanBase + '/m3u8/index.m3u8'; } if (!finalStreamUrl.startsWith('http://') && !finalStreamUrl.startsWith('https://')) { if (finalStreamUrl.startsWith('/')) { finalStreamUrl = 'https://main.24playerhd.com' + finalStreamUrl; } else { finalStreamUrl = 'https://main.24playerhd.com/' + finalStreamUrl; } } let proxyUrl = proxyBase + encodeURIComponent(finalStreamUrl) + '&referer=' + encodeURIComponent('https://main.24playerhd.com/'); if (Hls.isSupported()) { const hls = new Hls(); hls.loadSource(proxyUrl); hls.attachMedia(video); hls.on(Hls.Events.MANIFEST_PARSED, function() { video.play().catch(e => console.log("Play blocked:", e)); }); hls.on(Hls.Events.ERROR, function (event, data) { if (data.fatal) { switch (data.type) { case Hls.ErrorTypes.NETWORK_ERROR: hls.startLoad(); break; case Hls.ErrorTypes.MEDIA_ERROR: hls.recoverMediaError(); break; default: hls.destroy(); break; } } }); currentHls = hls; } else if (video.canPlayType('application/vnd.apple.mpegurl')) { video.src = proxyUrl; video.addEventListener('loadedmetadata', function() { video.play().catch(e => console.log("Play blocked:", e)); }); } }
